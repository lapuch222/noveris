
import 'dart:async';

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

// ============================================================
// ANHANGÁ — CONTROLE DO QUARTO VIRTUAL
//
// Abas:
// 1. Luz
// 2. Ambiente
// 3. Jogador
//
// Comunicação com o HTML:
// Canal: quarto:CODIGO
// Evento: estado
//
// Os controles manuais são uma simulação.
// Nenhuma medição é inventada ou salva como real.
// ============================================================

class QuartoVirtualPage extends StatefulWidget {
  const QuartoVirtualPage({
    super.key,
    required this.sessoes,
  });

  final List<Map<String, dynamic>> sessoes;

  @override
  State<QuartoVirtualPage> createState() =>
      _QuartoVirtualPageState();
}

class _QuartoVirtualPageState
    extends State<QuartoVirtualPage> {
  static const destaque = Color(0xFF5FD4EC);
  static const fundo = Color(0xFF04090D);
  static const painel = Color(0xFF0A1720);

  final codigoController = TextEditingController();

  bool lampadaLigada = false;
  double intensidadeLuz = 70;

  // Condições simuladas pelo usuário.
  double temperatura = 24;
  double ruido = 40;
  double umidade = 50;

  String personagem = 'adulto';

  // Permite escolher uma sessão real do histórico
  // em vez de valores configurados manualmente.
  int? indiceHistorico;

  bool enviando = false;

  String mensagem =
      'Digite o código exibido no quarto virtual.';

  String get codigo {
    return codigoController.text
        .trim()
        .toUpperCase()
        .replaceAll(
          RegExp(r'[^A-Z0-9]'),
          '',
        );
  }

  List<Map<String, dynamic>> get sessoesComAmbiente {
    return widget.sessoes.where((sessao) {
      return sessao['temperatura'] != null ||
          sessao['umidade'] != null ||
          sessao['ruido_db'] != null;
    }).toList();
  }

  double? converterNumero(Object? valor) {
    if (valor == null) return null;

    if (valor is num) {
      return valor.toDouble();
    }

    return double.tryParse(
      valor.toString().replaceAll(',', '.'),
    );
  }

  String nomePersonagem(String valor) {
    switch (valor) {
      case 'crianca':
        return 'Criança';
      case 'adolescente':
        return 'Adolescente';
      default:
        return 'Adulto';
    }
  }

  String situacaoTemperatura() {
    if (temperatura < 19) {
      return 'Está bem frio!';
    }

    if (temperatura < 22) {
      return 'Um pouco frio';
    }

    if (temperatura < 27) {
      return 'Temperatura agradável';
    }

    if (temperatura < 30) {
      return 'Está ficando quente';
    }

    return 'Muito quente!';
  }

  String situacaoRuido() {
    if (ruido < 20) {
      return 'Quarto bem silencioso';
    }

    if (ruido < 45) {
      return 'Ambiente tranquilo';
    }

    if (ruido < 70) {
      return 'Barulho moderado';
    }

    if (ruido < 85) {
      return 'Ambiente barulhento';
    }

    return 'Barulho muito alto!';
  }

  @override
  void dispose() {
    codigoController.dispose();
    super.dispose();
  }

  // ==========================================================
  // ENVIAR CONTROLES PARA O QUARTO
  // ==========================================================

  Future<void> enviarAoQuarto() async {
    if (enviando) return;

    final codigoAtual = codigo;

    if (codigoAtual.length != 6) {
      setState(() {
        mensagem =
            'Digite os 6 caracteres do código do quarto.';
      });
      return;
    }

    setState(() {
      enviando = true;
      mensagem = 'Conectando ao quarto...';
    });

    RealtimeChannel? canal;

    try {
      final supabase = Supabase.instance.client;

      canal = supabase.channel(
        'quarto:$codigoAtual',
      );

      final conectado = Completer<void>();

      canal.subscribe((status, error) {
        if (conectado.isCompleted) return;

        if (status ==
            RealtimeSubscribeStatus.subscribed) {
          conectado.complete();
        } else if (status ==
                RealtimeSubscribeStatus.channelError ||
            status ==
                RealtimeSubscribeStatus.timedOut ||
            status ==
                RealtimeSubscribeStatus.closed) {
          conectado.completeError(
            error ??
                Exception(
                  'Não foi possível conectar.',
                ),
          );
        }
      });

      await conectado.future.timeout(
        const Duration(seconds: 12),
      );

      final historico = sessoesComAmbiente;

      Map<String, dynamic>? sessaoEscolhida;

      if (indiceHistorico != null &&
          indiceHistorico! >= 0 &&
          indiceHistorico! < historico.length) {
        sessaoEscolhida = historico[indiceHistorico!];
      }

      final dados = <String, dynamic>{
        'pessoa': personagem,

        'lampada_ligada': lampadaLigada,

        'intensidade_luz':
            lampadaLigada ? intensidadeLuz.round() : 0,

        // Com histórico selecionado, usa somente
        // valores realmente salvos no banco.
        // Caso contrário, usa a simulação manual.
        'temperatura': sessaoEscolhida == null
            ? temperatura
            : converterNumero(
                sessaoEscolhida['temperatura'],
              ),

        'umidade': sessaoEscolhida == null
            ? umidade
            : converterNumero(
                sessaoEscolhida['umidade'],
              ),

        'ruido_db': sessaoEscolhida == null
            ? ruido
            : converterNumero(
                sessaoEscolhida['ruido_db'],
              ),
      };

      await canal.sendBroadcastMessage(
        event: 'estado',
        payload: dados,
      );

      if (!mounted) return;

      setState(() {
        mensagem =
            'Comando enviado! Veja as mudanças no quarto.';
      });
    } on TimeoutException {
      if (mounted) {
        setState(() {
          mensagem =
              'A conexão demorou muito. '
              'Confira a internet e o código.';
        });
      }
    } catch (e) {
      if (mounted) {
        setState(() {
          mensagem =
              'Não foi possível enviar: $e';
        });
      }
    } finally {
      if (canal != null) {
        await Supabase.instance.client
            .removeChannel(canal);
      }

      if (mounted) {
        setState(() {
          enviando = false;
        });
      }
    }
  }

  // ==========================================================
  // COMPONENTES VISUAIS
  // ==========================================================

  Widget cartao(
    String titulo,
    Widget conteudo,
  ) {
    return Container(
      width: double.infinity,
      padding: const EdgeInsets.all(18),
      margin: const EdgeInsets.only(bottom: 16),
      decoration: BoxDecoration(
        color: painel,
        borderRadius: BorderRadius.circular(16),
        border: Border.all(
          color: const Color(0xFF1B3B4A),
        ),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            titulo,
            style: const TextStyle(
              color: destaque,
              fontSize: 20,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 14),
          conteudo,
        ],
      ),
    );
  }

  Widget controleDeslizante({
    required String titulo,
    required String unidade,
    required double valor,
    required double minimo,
    required double maximo,
    required ValueChanged<double> aoAlterar,
  }) {
    return Column(
      children: [
        Row(
          mainAxisAlignment:
              MainAxisAlignment.spaceBetween,
          children: [
            Expanded(
              child: Text(
                titulo,
                style: const TextStyle(
                  fontSize: 16,
                ),
              ),
            ),
            Text(
              '${valor.round()}$unidade',
              style: const TextStyle(
                color: destaque,
                fontSize: 18,
                fontWeight: FontWeight.bold,
              ),
            ),
          ],
        ),
        Slider(
          value: valor,
          min: minimo,
          max: maximo,
          activeColor: destaque,
          onChanged: aoAlterar,
        ),
        Row(
          mainAxisAlignment:
              MainAxisAlignment.spaceBetween,
          children: [
            Text(
              '${minimo.round()}$unidade',
              style: const TextStyle(
                color: Colors.white54,
                fontSize: 12,
              ),
            ),
            Text(
              '${maximo.round()}$unidade',
              style: const TextStyle(
                color: Colors.white54,
                fontSize: 12,
              ),
            ),
          ],
        ),
      ],
    );
  }

  // ==========================================================
  // ABA 1 — LUZ
  // ==========================================================

  Widget abaLuz() {
    return ListView(
      padding: const EdgeInsets.all(16),
      children: [
        cartao(
          'Iluminação do quarto',
          Column(
            children: [
              SwitchListTile(
                contentPadding: EdgeInsets.zero,
                value: lampadaLigada,
                activeThumbColor: destaque,
                onChanged: (valor) {
                  setState(() {
                    lampadaLigada = valor;
                  });
                },
                secondary: Icon(
                  lampadaLigada
                      ? Icons.lightbulb
                      : Icons.lightbulb_outline,
                  color: destaque,
                  size: 32,
                ),
                title: const Text(
                  'Lâmpada virtual',
                ),
                subtitle: Text(
                  lampadaLigada
                      ? 'Ligada'
                      : 'Desligada',
                ),
              ),

              const SizedBox(height: 20),

              controleDeslizante(
                titulo: 'Brilho da lâmpada',
                unidade: '%',
                valor: intensidadeLuz,
                minimo: 0,
                maximo: 100,
                aoAlterar: (valor) {
                  setState(() {
                    intensidadeLuz = valor;
                  });
                },
              ),

              const SizedBox(height: 8),

              const Text(
                'Altere o brilho para deixar '
                'o quarto virtual mais claro ou escuro.',
                style: TextStyle(
                  color: Colors.white70,
                ),
              ),
            ],
          ),
        ),
      ],
    );
  }

  // ==========================================================
  // ABA 2 — AMBIENTE
  // ==========================================================

  Widget abaAmbiente() {
    return ListView(
      padding: const EdgeInsets.all(16),
      children: [
        cartao(
          'Temperatura do quarto',
          Column(
            crossAxisAlignment:
                CrossAxisAlignment.start,
            children: [
              const Text(
                'Escolha a temperatura que deseja '
                'simular no quarto gamer.',
              ),

              const SizedBox(height: 22),

              controleDeslizante(
                titulo: 'Temperatura',
                unidade: ' °C',
                valor: temperatura,
                minimo: 16,
                maximo: 35,
                aoAlterar: (valor) {
                  setState(() {
                    temperatura = valor;
                    indiceHistorico = null;
                  });
                },
              ),

              const SizedBox(height: 12),

              Row(
                children: [
                  const Icon(
                    Icons.thermostat,
                    color: destaque,
                  ),
                  const SizedBox(width: 10),
                  Expanded(
                    child: Text(
                      situacaoTemperatura(),
                    ),
                  ),
                ],
              ),
            ],
          ),
        ),

        cartao(
          'Barulho do quarto',
          Column(
            crossAxisAlignment:
                CrossAxisAlignment.start,
            children: [
              const Text(
                'Simule um ambiente silencioso '
                'ou com mais barulho.',
              ),

              const SizedBox(height: 22),

              controleDeslizante(
                titulo: 'Nível de ruído',
                unidade: ' dB',
                valor: ruido,
                minimo: 0,
                maximo: 110,
                aoAlterar: (valor) {
                  setState(() {
                    ruido = valor;
                    indiceHistorico = null;
                  });
                },
              ),

              const SizedBox(height: 12),

              Row(
                children: [
                  const Icon(
                    Icons.graphic_eq,
                    color: destaque,
                  ),
                  const SizedBox(width: 10),
                  Expanded(
                    child: Text(
                      situacaoRuido(),
                    ),
                  ),
                ],
              ),
            ],
          ),
        ),

        cartao(
          'Umidade do ar',
          Column(
            children: [
              controleDeslizante(
                titulo: 'Umidade',
                unidade: '%',
                valor: umidade,
                minimo: 0,
                maximo: 100,
                aoAlterar: (valor) {
                  setState(() {
                    umidade = valor;
                    indiceHistorico = null;
                  });
                },
              ),

              const SizedBox(height: 12),

              const Text(
                'A umidade também pode mudar '
                'os efeitos visuais do quarto.',
                style: TextStyle(
                  color: Colors.white70,
                ),
              ),
            ],
          ),
        ),

        cartao(
          'Usar dados de uma partida',
          Column(
            crossAxisAlignment:
                CrossAxisAlignment.start,
            children: [
              const Text(
                'Opcional: em vez dos valores '
                'simulados acima, escolha uma '
                'partida com dados ambientais salvos.',
              ),

              const SizedBox(height: 14),

              DropdownButtonFormField<int>(
                key: ValueKey(indiceHistorico),
                initialValue: indiceHistorico,
                isExpanded: true,
                decoration: const InputDecoration(
                  labelText: 'Partida registrada',
                  border: OutlineInputBorder(),
                ),
                items: [
                  const DropdownMenuItem<int>(
                    value: -1,
                    child: Text(
                      'Usar controles manuais',
                    ),
                  ),
                  for (int i = 0;
                      i < sessoesComAmbiente.length;
                      i++)
                    DropdownMenuItem<int>(
                      value: i,
                      child: Text(
                        'Partida '
                        '${sessoesComAmbiente[i]['id_fato']}',
                      ),
                    ),
                ],
                onChanged: (indice) {
                  setState(() {
                    indiceHistorico =
                        indice == null || indice == -1
                            ? null
                            : indice;
                  });
                },
              ),

              if (indiceHistorico != null)
                const Padding(
                  padding: EdgeInsets.only(top: 12),
                  child: Text(
                    'Os valores enviados serão os '
                    'da partida selecionada.',
                    style: TextStyle(
                      color: destaque,
                    ),
                  ),
                ),

              const SizedBox(height: 12),

              const Text(
                'Os controles manuais representam '
                'uma simulação, não medições reais.',
                style: TextStyle(
                  color: Colors.white70,
                  fontSize: 12,
                ),
              ),
            ],
          ),
        ),
      ],
    );
  }

  // ==========================================================
  // ABA 3 — JOGADOR
  // ==========================================================

  Widget abaJogador() {
    return ListView(
      padding: const EdgeInsets.all(16),
      children: [
        cartao(
          'Quem está jogando?',
          Column(
            crossAxisAlignment:
                CrossAxisAlignment.start,
            children: [
              const Text(
                'Escolha qual personagem aparece '
                'na cadeira do quarto virtual.',
              ),

              const SizedBox(height: 18),

              RadioGroup<String>(
                groupValue: personagem,
                onChanged: (valor) {
                  if (valor != null) {
                    setState(() {
                      personagem = valor;
                    });
                  }
                },
                child: const Column(
                  children: [
                    RadioListTile<String>(
                      title: Text('Criança'),
                      secondary: Icon(
                        Icons.child_care,
                      ),
                      value: 'crianca',
                    ),
                    RadioListTile<String>(
                      title: Text('Adolescente'),
                      secondary: Icon(
                        Icons.person_outline,
                      ),
                      value: 'adolescente',
                    ),
                    RadioListTile<String>(
                      title: Text('Adulto'),
                      secondary: Icon(
                        Icons.person,
                      ),
                      value: 'adulto',
                    ),
                  ],
                ),
              ),

              const SizedBox(height: 12),

              Text(
                'Personagem escolhido: '
                '${nomePersonagem(personagem)}',
                style: const TextStyle(
                  color: destaque,
                  fontWeight: FontWeight.bold,
                ),
              ),
            ],
          ),
        ),
      ],
    );
  }

  // ==========================================================
  // INTERFACE PRINCIPAL
  // ==========================================================

  @override
  Widget build(BuildContext context) {
    return DefaultTabController(
      length: 3,
      child: Scaffold(
        backgroundColor: fundo,

        appBar: AppBar(
          backgroundColor:
              const Color(0xFF08151D),
          title: const Text(
            'Meu quarto virtual',
          ),
          bottom: const TabBar(
            indicatorColor: destaque,
            labelColor: destaque,
            tabs: [
              Tab(
                icon: Icon(Icons.lightbulb_outline),
                text: 'Luz',
              ),
              Tab(
                icon: Icon(Icons.thermostat),
                text: 'Ambiente',
              ),
              Tab(
                icon: Icon(Icons.person_outline),
                text: 'Jogador',
              ),
            ],
          ),
        ),

        body: Column(
          children: [
            Padding(
              padding:
                  const EdgeInsets.fromLTRB(
                16,
                16,
                16,
                0,
              ),
              child: TextField(
                controller: codigoController,
                maxLength: 6,
                textCapitalization:
                    TextCapitalization.characters,
                inputFormatters: [
                  FilteringTextInputFormatter.allow(
                    RegExp(r'[a-zA-Z0-9]'),
                  ),
                ],
                decoration: const InputDecoration(
                  labelText:
                      'Código do quarto',
                  hintText: 'Ex.: ABC123',
                  prefixIcon: Icon(
                    Icons.link,
                  ),
                  border:
                      OutlineInputBorder(),
                ),
              ),
            ),

            Expanded(
              child: TabBarView(
                children: [
                  abaLuz(),
                  abaAmbiente(),
                  abaJogador(),
                ],
              ),
            ),

            Container(
              width: double.infinity,
              padding: const EdgeInsets.all(16),
              decoration: const BoxDecoration(
                color: Color(0xFF08151D),
                border: Border(
                  top: BorderSide(
                    color: Color(0xFF1B3B4A),
                  ),
                ),
              ),
              child: Column(
                children: [
                  FilledButton.icon(
                    onPressed:
                        enviando ? null : enviarAoQuarto,
                    icon: const Icon(
                      Icons.send,
                    ),
                    label: Text(
                      enviando
                          ? 'Enviando...'
                          : 'Atualizar meu quarto',
                    ),
                  ),

                  const SizedBox(height: 10),

                  Text(
                    mensagem,
                    textAlign: TextAlign.center,
                    style: const TextStyle(
                      color: destaque,
                      fontSize: 12,
                    ),
                  ),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }
}
